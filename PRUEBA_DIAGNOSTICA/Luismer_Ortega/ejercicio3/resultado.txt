#include <stdlib.h>

typedef struct Nodo {
    int valor;
    struct Nodo *siguiente;
} Nodo;

int factorial(int n) {
    if (n <= 0) return 1;
    return n * factorial(n - 1);
}

int main(void) {
    int i;
    float suma = 0.0;
    char letra = 'A';
    const int MAX = 10;
    unsigned int contador = 0;

    for (i = 0; i < MAX; i++) {
        if (i % 2 == 0) {
            suma += (float)i;
            contador++;
        } else {
            continue;
        }
    }

    while (contador > 0) {
        switch (contador % 3) {
            case 0: break;
            case 1: suma -= 1.0; break;
            default: suma += 0.5;
        }
        contador--;
    }

    int *arr = (int *)malloc(MAX * sizeof(int));
    if (arr == NULL) return 1;

    for (i = 0; i < MAX; i++) {
        arr[i] = factorial(i);
    }

    free(arr);
    return 0;
}
